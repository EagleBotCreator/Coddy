cmake_minimum_required(VERSION 3.22)
project(coddy LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Keep inference kernels optimized in the personal-use debug APK as well.
set(CMAKE_C_FLAGS_DEBUG "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
 set(GGML_AVX OFF CACHE BOOL "" FORCE)
 set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
 set(GGML_FMA OFF CACHE BOOL "" FORCE)
 set(GGML_F16C OFF CACHE BOOL "" FORCE)
endif()
if(ANDROID_ABI STREQUAL "arm64-v8a")
 set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
include(FetchContent)
FetchContent_Declare(llama
 URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/fc504f91d8cfdfcd9b0fedcf14600fd2b0e63420
 URL_HASH SHA256=2ec592bdc0d45faed362c6ccd8c8208b53ebb46811cd131292a1688ce501b7fe
)
FetchContent_MakeAvailable(llama)
add_library(coddy SHARED coddy_jni.cpp)
target_link_libraries(coddy PRIVATE llama android log)
